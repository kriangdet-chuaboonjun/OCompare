cmake_minimum_required(VERSION 3.21)
project(OCompareNative LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
enable_testing()
set(OCOMPARE_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../src-tauri/icons/icon.ico")
configure_file(explorer-extension/resources.rc.in "${CMAKE_CURRENT_BINARY_DIR}/resources.rc" @ONLY)
add_library(OCompareCore STATIC explorer-extension/selection.cpp explorer-extension/launch.cpp)
target_include_directories(OCompareCore PUBLIC explorer-extension)
target_link_libraries(OCompareCore PUBLIC advapi32 shlwapi shell32 ole32)
add_library(OCompareExplorer SHARED explorer-extension/command.cpp explorer-extension/module.cpp explorer-extension/exports.def "${CMAKE_CURRENT_BINARY_DIR}/resources.rc")
target_link_libraries(OCompareExplorer PRIVATE OCompareCore)
set_target_properties(OCompareExplorer PROPERTIES OUTPUT_NAME "OCompare.Explorer")
add_executable(OCompareNativeTests tests/nativeTests.cpp)
target_link_libraries(OCompareNativeTests PRIVATE OCompareCore)
set_target_properties(OCompareNativeTests PROPERTIES OUTPUT_NAME "OCompare.NativeTests")
add_test(NAME native-contracts COMMAND OCompareNativeTests)
add_dependencies(OCompareNativeTests OCompareExplorer)
set(OCOMPARE_PUBLISHER "$ENV{OCOMPARE_PACKAGE_PUBLISHER}")
if(NOT OCOMPARE_PUBLISHER)
  set(OCOMPARE_PUBLISHER "CN=FusionSoft")
endif()
string(REPLACE "&" "&amp;" OCOMPARE_PUBLISHER "${OCOMPARE_PUBLISHER}")
string(REPLACE "\"" "&quot;" OCOMPARE_PUBLISHER "${OCOMPARE_PUBLISHER}")
configure_file(launcher/identity.manifest.in "${CMAKE_CURRENT_BINARY_DIR}/launcher.manifest" @ONLY)
configure_file(launcher/resources.rc.in "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc" @ONLY)
add_executable(OCompareLauncher WIN32 launcher/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc")
target_link_libraries(OCompareLauncher PRIVATE OCompareCore user32)
target_link_options(OCompareLauncher PRIVATE "/MANIFEST:NO")
set_target_properties(OCompareLauncher PROPERTIES OUTPUT_NAME "OCompare.Launcher")
